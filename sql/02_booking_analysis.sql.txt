USE TaxiAnalytics;

SELECT
    Booking_Status,
    COUNT(*) AS Booking_Count,
    CAST(
        COUNT(*) * 100.0 / SUM(COUNT(*)) OVER()
        AS DECIMAL(10,2)
    ) AS Booking_Percentage
FROM dbo.Bookings_Cleaned
GROUP BY Booking_Status
ORDER BY Booking_Count DESC;

SELECT
    Payment_Method,
    COUNT(*) AS Booking_Count,
    SUM(Booking_Value) AS Total_Revenue
FROM dbo.Bookings_Cleaned
WHERE Payment_Method IS NOT NULL
GROUP BY Payment_Method
ORDER BY Total_Revenue DESC;

SELECT
    Vehicle_Type,
    COUNT(*) AS Total_Bookings,
    SUM(Booking_Value) AS Total_Revenue,
    AVG(CAST(Booking_Value AS DECIMAL(10,2))) AS Average_Fare,
    AVG(CAST(Ride_Distance AS DECIMAL(10,2))) AS Average_Ride_Distance
FROM dbo.Bookings_Cleaned
GROUP BY Vehicle_Type
ORDER BY Total_Bookings DESC;

SELECT
    CAST(DateTime AS DATE) AS Booking_Date,
    COUNT(*) AS Total_Bookings,
    SUM(Booking_Value) AS Total_Revenue
FROM dbo.Bookings_Cleaned
GROUP BY CAST(DateTime AS DATE)
ORDER BY Booking_Date;

SELECT
    DATEPART(HOUR, DateTime) AS Booking_Hour,
    COUNT(*) AS Total_Bookings
FROM dbo.Bookings_Cleaned
GROUP BY DATEPART(HOUR, DateTime)
ORDER BY Booking_Hour;
